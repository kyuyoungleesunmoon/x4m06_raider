━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

⭐ 가장 간단한 실행 방법

   1. X4M06 레이더를 USB 포트에 연결
   2. run_x4m06.bat 파일을 더블클릭
   3. 메뉴에서 "1. 간편 데이터 수집기" 선택
   4. 원하는 수집 모드 선택
   5. 데이터 수집 완료!

🆘 문제 발생 시

   1. README_사용가이드.md의 "문제 해결" 섹션 확인
   2. python quick_test.py로 연결 상태 확인
   3. conda activate x4m06_env 환경 활성화 확인

✅ 필수 준비 사항

   • X4M06 레이더 하드웨어 (USB 연결)
   • Python 3.6 가상환경 (x4m06_env)
   • pymoduleconnector 라이브러리 설치됨
   • COM 포트 연결 (보통 COM3)
